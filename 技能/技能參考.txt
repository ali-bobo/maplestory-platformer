class Shuriken {
  constructor(x, y, direction) {
    this.x = x;
    this.y = y;
    this.speed = 12;
    this.direction = direction;
    this.rotation = 0;
    this.active = true;
  }

  update() {
    this.x += this.speed * this.direction;
    this.rotation += 0.3;

    // 簡單碰撞判定（你可換成引擎版）
    if (this.hitEnemy()) {
      this.onHit();
    }
  }

  draw(ctx, sprite) {
    ctx.save();
    ctx.translate(this.x, this.y);
    ctx.rotate(this.rotation);
    
    // 拖尾（簡單做法）
    ctx.globalAlpha = 0.3;
    ctx.drawImage(sprite, -20, -20, 40, 40);
    ctx.globalAlpha = 1;

    ctx.drawImage(sprite, -16, -16, 32, 32);
    ctx.restore();
  }

  hitEnemy() {
    // TODO: 你的碰撞邏輯
    return false;
  }

  onHit() {
    this.active = false;

    // 👉 產生命中特效
    spawnHitEffect(this.x, this.y);

    // 👉 停頓（hit stop，讓手感變好）
    game.freeze(0.05);
  }
}


game.freeze = function(duration) {
  this.freezeTime = duration;
};
