我在用 Phaser 3 製作一個2D動作遊戲，幫我改寫以下技能函式 castShuriken。
需求：
1. 每枚飛鏢後方加入紫藍色尾跡粒子（使用 scene.add.particles，每 20ms emit 一個，生命 300ms，顏色 0x8888ff→0x4400ff 漸退）
2. 命中爆炸改為扇形 8 粒子噴射（而非單點 spawnHit）
3. 玩家 sprite 需播放 'throw' animation（持續 200ms 後自動 stop）
4. 飛鏢 sprite 加入旋轉拖影（每 50ms 生成半透明殘影，alpha 0.4→0，持續 150ms）

 點擊複製我在用 Phaser 3 製作一個2D動作遊戲，幫我改寫 castDash 技能函式。
需求：
1. 衝刺起點：生成 1 個白色衝擊圓（scene.add.graphics 圓形，半徑 30→80，alpha 0.8→0，duration 250ms）
2. 殘影系統：改為每 25ms 生成 1 個玩家當前 sprite 的複製 image（setTint(0x8800ff), alpha 0.6→0, duration 300ms，最多同時存在 6 個）
3. 衝刺期間玩家閃爍（scene.tweens.add alpha 1→0.3→1，yoyo: true，repeat: 3，duration: 80ms 每次）
4. 衝刺結束時生成落地粒子（地面位置 10 個灰紫色粒子向外噴射）

我在用 Phaser 3 製作一個2D動作遊戲，幫我改寫 castAssassinate 技能函式。
需求：
1. 瞬移前：全螢幕白色閃光（scene.cameras.main.flash(80, 255,255,255)）
2. 瞬移後玩家出現位置：生成暗紫色衝擊波圓（半徑 0→120，alpha 0.7→0，lineStyle 3px 0xff00ff，duration 350ms）
3. 敵人被擊中後：加入昏迷 icon（scene.add.text 在敵人頭頂顯示 '★' 旋轉，持續 2000ms 後 fadeOut）
4. 保留現有 burst 放射線效果，但加入 scene.cameras.main.shake(200, 0.025) 強化震動
5. 敵人死亡時 tween scaleX/Y 0.1 + alpha 0（duration 400ms）而非立即消失

我在用 Phaser 3 製作一個2D動作遊戲，幫我改寫 castVortex 技能函式。
需求：
1. 魔法陣改為旋轉動畫：使用 scene.tweens.add 讓 circle 物件 angle 從 0→360，duration: 1200ms
2. AoE 範圍提示：先顯示半透明紅圈（radius 180，alpha 0.15，fillStyle 0xff0000）持續 300ms 再開始施放
3. 每個能量球加入尾跡：使用 Phaser.GameObjects.Particles，emitZone 跟隨 orb，粒子顏色 0x9933ff，lifespan 200ms
4. 所有 8 顆能量球同時發射（移除 delayedCall 的間隔），改為視覺上從魔法陣邊緣「彈出」動畫（tween scale 0→1，duration 150ms）
5. 技能結束時圓圈向外爆炸消散（scale 1→2.5，alpha 0.8→0，duration 400ms）

我在用 Phaser 3 製作一個2D動作遊戲，幫我改寫 castClone 技能函式。
需求：
1. 召喚時：在 clone 位置播放召喚特效（紫色粒子從地面向上噴發，count 20，vy -200→-50，持續 400ms）
2. 分身存在期間：持續微閃（scene.tweens.add alpha 0.7→0.4，yoyo:true，repeat:-1，duration:600ms）
3. 分身頭頂顯示生命條（scene.add.graphics，寬 40px，隨時間從綠→黃→紅，5 秒倒數）
4. 分身攻擊時：播放揮擊特效（短暫白色弧線，scene.add.graphics，alpha 1→0，duration 150ms）
5. 消散時：改為向上飄散（clone.y 往上 tween -60px + alpha 0，同時生成 15 個紫色粒子向四周散射）

<script>
// ── Skill 1: Dark energy slash (Swan-like)
const c1 = document.getElementById('c1');
const ctx1 = c1.getContext('2d');
let particles1 = [];
let slashAngle = 0, slashProgress = 0, slashActive = false;
let animId1 = null;

function triggerSkill1() {
  document.getElementById('btn1').className = 'skill-btn active';
  document.getElementById('btn2').className = 'skill-btn';
  particles1 = [];
  slashProgress = 0;
  slashActive = true;
  for (let i = 0; i < 60; i++) {
    particles1.push({
      x: 150 + (Math.random()-0.5)*60,
      y: 110 + (Math.random()-0.5)*40,
      vx: (Math.random()-0.5)*6,
      vy: -(Math.random()*5+1),
      life: 1,
      decay: 0.012 + Math.random()*0.018,
      size: Math.random()*3+1,
      hue: 280 + Math.random()*60
    });
  }
  if (!animId1) loop1();
}

function loop1() {
  ctx1.clearRect(0, 0, 300, 220);
  ctx1.fillStyle = 'rgba(5,5,16,0.3)';
  ctx1.fillRect(0, 0, 300, 220);

  if (slashActive) {
    slashProgress = Math.min(slashProgress + 0.05, 1);
    // Draw arc slash
    const cx = 150, cy = 110;
    ctx1.save();
    const len = slashProgress;
    const startA = -2.2, endA = startA + len * 3.6;
    const r1 = 30, r2 = 90;
    ctx1.beginPath();
    ctx1.arc(cx, cy, r2, startA, endA);
    ctx1.arc(cx, cy, r1, endA, startA, true);
    ctx1.closePath();
    const grd = ctx1.createLinearGradient(cx-r2, cy, cx+r2, cy);
    grd.addColorStop(0, 'rgba(180,80,255,0)');
    grd.addColorStop(0.5, `rgba(220,100,255,${0.6*slashProgress})`);
    grd.addColorStop(1, 'rgba(255,150,255,0)');
    ctx1.fillStyle = grd;
    ctx1.fill();

    // Slash bright edge
    ctx1.beginPath();
    ctx1.arc(cx, cy, r2+4, startA, endA);
    ctx1.strokeStyle = `rgba(255,200,255,${0.8*slashProgress})`;
    ctx1.lineWidth = 2;
    ctx1.stroke();

    // Feather streaks
    for (let i = 0; i < 8; i++) {
      const a = startA + (endA-startA)*(i/8);
      const sx = cx + Math.cos(a)*r2;
      const sy = cy + Math.sin(a)*r2;
      const ex = sx + Math.cos(a-0.5)*(20+i*4);
      const ey = sy + Math.sin(a-0.5)*(20+i*4);
      ctx1.beginPath();
      ctx1.moveTo(sx, sy);
      ctx1.lineTo(ex, ey);
      ctx1.strokeStyle = `rgba(240,180,255,${0.4*slashProgress})`;
      ctx1.lineWidth = 1;
      ctx1.stroke();
    }
    if (slashProgress >= 1) slashActive = false;
    ctx1.restore();
  }

  // Particles
  particles1 = particles1.filter(p => p.life > 0);
  for (const p of particles1) {
    p.x += p.vx; p.y += p.vy;
    p.vy += 0.05;
    p.vx *= 0.97;
    p.life -= p.decay;
    ctx1.beginPath();
    ctx1.arc(p.x, p.y, p.size * p.life, 0, Math.PI*2);
    ctx1.fillStyle = `hsla(${p.hue},90%,75%,${p.life*0.8})`;
    ctx1.fill();
  }

  if (particles1.length > 0 || slashActive) {
    animId1 = requestAnimationFrame(loop1);
  } else {
    animId1 = null;
    ctx1.clearRect(0,0,300,220);
  }
}

// ── Skill 2: Golden magic circle
const c2 = document.getElementById('c2');
const ctx2 = c2.getContext('2d');
let circleActive = false, circleT = 0;
let sparks = [];
let animId2 = null;
let pulseScale = 0, glowAlpha = 0;

function triggerSkill2() {
  document.getElementById('btn2').className = 'skill-btn active';
  document.getElementById('btn1').className = 'skill-btn';
  circleActive = true;
  circleT = 0;
  sparks = [];
  pulseScale = 0;
  glowAlpha = 0;
  for (let i = 0; i < 40; i++) {
    const a = Math.random()*Math.PI*2;
    sparks.push({
      x: 150, y: 110,
      vx: Math.cos(a)*(2+Math.random()*4),
      vy: Math.sin(a)*(2+Math.random()*4),
      life: 1, decay: 0.015+Math.random()*0.02,
      size: 1+Math.random()*2.5
    });
  }
  if (!animId2) loop2();
}

function loop2() {
  ctx2.clearRect(0, 0, 300, 220);
  ctx2.fillStyle = 'rgba(255,248,220,0.05)';
  ctx2.fillRect(0,0,300,220);

  const cx=150, cy=110;
  circleT += 0.04;
  if (circleActive) {
    pulseScale = Math.min(pulseScale+0.04, 1);
    glowAlpha = Math.min(glowAlpha+0.05, 0.9);
  } else {
    glowAlpha = Math.max(glowAlpha-0.02, 0);
    pulseScale = Math.max(pulseScale-0.02, 0);
  }

  const r = 65 * pulseScale;
  if (r > 2) {
    // Outer glow
    const glow = ctx2.createRadialGradient(cx,cy,r*0.5,cx,cy,r*1.4);
    glow.addColorStop(0, `rgba(255,215,0,${glowAlpha*0.3})`);
    glow.addColorStop(1, 'rgba(255,200,0,0)');
    ctx2.fillStyle = glow;
    ctx2.beginPath(); ctx2.arc(cx,cy,r*1.4,0,Math.PI*2); ctx2.fill();

    // Main ring
    ctx2.beginPath();
    ctx2.arc(cx, cy, r, 0, Math.PI*2);
    ctx2.strokeStyle = `rgba(255,215,0,${glowAlpha})`;
    ctx2.lineWidth = 3;
    ctx2.stroke();

    // Inner ring
    ctx2.beginPath();
    ctx2.arc(cx, cy, r*0.65, 0, Math.PI*2);
    ctx2.strokeStyle = `rgba(255,200,0,${glowAlpha*0.7})`;
    ctx2.lineWidth = 1.5;
    ctx2.stroke();

    // Rotating runes (8 points)
    for (let i = 0; i < 8; i++) {
      const a = (i/8)*Math.PI*2 + circleT;
      const px = cx + Math.cos(a)*r;
      const py = cy + Math.sin(a)*r;
      ctx2.beginPath();
      ctx2.arc(px, py, 4*pulseScale, 0, Math.PI*2);
      ctx2.fillStyle = `rgba(255,240,100,${glowAlpha})`;
      ctx2.fill();
      // Spoke
      ctx2.beginPath();
      ctx2.moveTo(cx + Math.cos(a)*r*0.65, cy + Math.sin(a)*r*0.65);
      ctx2.lineTo(px, py);
      ctx2.strokeStyle = `rgba(255,220,50,${glowAlpha*0.5})`;
      ctx2.lineWidth = 1;
      ctx2.stroke();
    }

    // Inner cross detail
    ctx2.save();
    ctx2.translate(cx,cy);
    ctx2.rotate(circleT*0.5);
    ctx2.strokeStyle = `rgba(255,215,0,${glowAlpha*0.6})`;
    ctx2.lineWidth = 1;
    for (let i=0;i<4;i++){
      const a2=(i/4)*Math.PI*2;
      ctx2.beginPath();
      ctx2.moveTo(0,0);
      ctx2.lineTo(Math.cos(a2)*r*0.5, Math.sin(a2)*r*0.5);
      ctx2.stroke();
    }
    ctx2.restore();

    // Starburst at edge points
    for (let i=0;i<4;i++){
      const a3 = (i/4)*Math.PI*2 + circleT*1.5;
      const sx = cx+Math.cos(a3)*r;
      const sy = cy+Math.sin(a3)*r;
      for (let j=0;j<4;j++){
        const sa=(j/4)*Math.PI*2;
        ctx2.beginPath();
        ctx2.moveTo(sx,sy);
        ctx2.lineTo(sx+Math.cos(sa)*8*pulseScale, sy+Math.sin(sa)*8*pulseScale);
        ctx2.strokeStyle=`rgba(255,255,200,${glowAlpha*0.8})`;
        ctx2.lineWidth=1.2;
        ctx2.stroke();
      }
    }
    if (circleT > 3) circleActive = false;
  }

  // Sparks
  sparks = sparks.filter(p=>p.life>0);
  for (const p of sparks){
    p.x+=p.vx; p.y+=p.vy;
    p.vx*=0.94; p.vy*=0.94;
    p.life-=p.decay;
    ctx2.beginPath();
    ctx2.arc(p.x,p.y,p.size*p.life,0,Math.PI*2);
    ctx2.fillStyle=`rgba(255,220,50,${p.life*0.9})`;
    ctx2.fill();
  }

  if (sparks.length>0 || circleActive || glowAlpha>0.01){
    animId2=requestAnimationFrame(loop2);
  } else {
    animId2=null;
    ctx2.clearRect(0,0,300,220);
  }
}

triggerSkill1();
setTimeout(()=>triggerSkill2(),400);
</script>
